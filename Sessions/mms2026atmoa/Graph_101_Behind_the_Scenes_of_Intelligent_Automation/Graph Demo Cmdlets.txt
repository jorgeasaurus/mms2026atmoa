#region Get Intune Policy, Settings, Export & Import
#Connect with Scope
#Connect-MgGraph -Scopes DeviceManagementConfiguration.Read.All
#Connect-MgGraph -Scopes DeviceManagementConfiguration.ReadWrite.All

$policyID = "<POLICY ID HERE>"
$invokeReqPolicy = Invoke-MgGraphRequest -Method GET -Uri "https://graph.microsoft.com/beta/deviceManagement/configurationPolicies('$policyID')"
$invokeReqPolicySettings = Invoke-MgGraphRequest -Method GET -Uri "https://graph.microsoft.com/beta/deviceManagement/configurationPolicies('$policyID')/settings"

#Paginate
$All_GraphRequest = @()
$All_GraphRequest += $invokeReqPolicySettings
while ($invokeReqPolicySettings.'@odata.nextLink')
{
    $invokeReqPolicySettings_NextLink = @{
        Method = "GET"
        URI    = $invokeReqPolicySettings.'@odata.nextLink'
    }
    $invokeReqPolicySettings = Invoke-MgGraphRequest @invokeReqPolicySettings_NextLink -ErrorAction Stop
    $All_GraphRequest += $invokeReqPolicySettings
}
$policySettings = $All_GraphRequest | Select-Object -ExpandProperty Value   

$exportPolicyValue = $invokeReqPolicy
$exportPolicyValue.settings = $policySettings

#Export JSON
$invokeReqPolicy | ConvertTo-Json -Depth 10 | Out-File "C:\Temp\$($invokeReqPolicy.name).json"

#Import JSON Policy
$jsonPolicy = Get-Content "C:\Temp\$($invokeReqPolicy.name).json" | ConvertFrom-Json
$jsonPolicy = $jsonPolicy | Select-Object -ExcludeProperty roleScopeTagIds, id, createdDateTime, lastModifiedDateTime, settingCount, creationSource, priorityMetaData
$jsonPolicy.name = $jsonPolicy.name + " - MMS MOA Import"

#Copy to Clipboard for Reference
$jsonPolicy | ConvertTo-Json -Depth 10 | Set-Clipboard

Invoke-MgGraphRequest POST -Uri "https://graph.microsoft.com/beta/deviceManagement/configurationPolicies" -Body ( $jsonPolicy | ConvertTo-Json -Depth 10 )


#endregion Get Intune Policy

#endregion Graph API Calls
